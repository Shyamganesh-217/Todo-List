import React, { useState } from "react";
 
/*
  ==========================================================
  TODO LIST - useState Hook Reference (Tanglish Explanation)
  ==========================================================
 
  useState Hook Basics:
  ----------------------
  const [state, setState] = useState(initialValue);
 
  - "state"      -> current value (naanga eppo padikkirom)
  - "setState"   -> function to UPDATE the state
  - "initialValue" -> first-time value (component mount aagum bodhu)
 
  IMPORTANT RULE: Neenga direct-a state variable-a mudhalikave mudiyathu.
  Example: todos.push(newTodo)  -> WRONG! React re-render aagathu.
  Always use setTodos([...todos, newTodo]) -> puthu array create pannunga.
 
  Idhu React oda "Immutability" concept - old state-a touch pannama,
  new copy create panni replace pannanum.
*/
 
function TodoList() {
  // 1) State for the list of todos (array of objects)
  const [todos, setTodos] = useState([]);
 
  // 2) State for the input box value (controlled component)
  const [inputValue, setInputValue] = useState("");
 
  // 3) State to track WHICH todo is currently being edited (id or null)
  const [editingId, setEditingId] = useState(null);
 
  // 4) State for the text inside the edit input box
  const [editText, setEditText] = useState("");
 
  // ---- ADD a new todo ----
  const handleAddTodo = () => {
    if (inputValue.trim() === "") return; // empty check
 
    const newTodo = {
      id: Date.now(),       // unique id (timestamp use pannalam)
      text: inputValue,
      completed: false,
    };
 
    // Spread operator (...todos) use panni old items keep pannitu
    // new item add pannurom -> puthu array reference varum -> React re-render aagum
    setTodos([...todos, newTodo]);
 
    setInputValue(""); // input box clear pannurom
  };
 
  // ---- TOGGLE complete/incomplete ----
  const handleToggle = (id) => {
    setTodos(
      todos.map((todo) =>
        todo.id === id ? { ...todo, completed: !todo.completed } : todo
      )
    );
    // map() use panni ORIGINAL array-a modify pannama, puthu array return pannurom
  };
 
  // ---- DELETE a todo ----
  const handleDelete = (id) => {
    setTodos(todos.filter((todo) => todo.id !== id));
    // filter() -> matching id illatha items mattum vachikittu puthu array
  };
 
  // ---- UPDATE: start editing a todo ----
  const handleEditStart = (todo) => {
    setEditingId(todo.id);   // idhu edit mode ON pannuthu, indha id ku mattum
    setEditText(todo.text);  // current text-a edit box la pre-fill pannurom
  };
 
  // ---- UPDATE: save the edited todo ----
  const handleEditSave = (id) => {
    if (editText.trim() === "") return; // empty text save pannadha
 
    setTodos(
      todos.map((todo) =>
        todo.id === id ? { ...todo, text: editText } : todo
      )
    );
    // map() use panni matching id oda text mattum update pannurom,
    // remaining fields (id, completed) same-a spread (...todo) vachikirom
 
    setEditingId(null); // edit mode OFF
    setEditText("");
  };
 
  // ---- UPDATE: cancel editing ----
  const handleEditCancel = () => {
    setEditingId(null);
    setEditText("");
  };
 
  // ---- UPDATE: Enter/Escape key support while editing ----
  const handleEditKeyDown = (e, id) => {
    if (e.key === "Enter") handleEditSave(id);
    if (e.key === "Escape") handleEditCancel();
  };
 
  // ---- EDIT input change handler ----
  const handleInputChange = (e) => {
    setInputValue(e.target.value);
  };
 
  // ---- ENTER key support ----
  const handleKeyDown = (e) => {
    if (e.key === "Enter") {
      handleAddTodo();
    }
  };
 
  const completedCount = todos.filter((t) => t.completed).length;
 
  return (
    <div style={styles.container}>
      <h2 style={styles.heading}>My Todo List (useState Demo)</h2>
 
      <div style={styles.inputRow}>
        <input
          type="text"
          value={inputValue}
          onChange={handleInputChange}
          onKeyDown={handleKeyDown}
          placeholder="Enter a task..."
          style={styles.input}
        />
        <button onClick={handleAddTodo} style={styles.addBtn}>
          Add
        </button>
      </div>
 
      {todos.length === 0 ? (
        <p style={styles.emptyText}>No tasks yet. Add one da!</p>
      ) : (
        <ul style={styles.list}>
          {todos.map((todo) => (
            <li key={todo.id} style={styles.listItem}>
              {editingId === todo.id ? (
                // ---- EDIT MODE: show input box instead of text ----
                <>
                  <input
                    type="text"
                    value={editText}
                    onChange={(e) => setEditText(e.target.value)}
                    onKeyDown={(e) => handleEditKeyDown(e, todo.id)}
                    autoFocus
                    style={styles.editInput}
                  />
                  <button
                    onClick={() => handleEditSave(todo.id)}
                    style={styles.saveBtn}
                  >
                    Save
                  </button>
                  <button onClick={handleEditCancel} style={styles.cancelBtn}>
                    Cancel
                  </button>
                </>
              ) : (
                // ---- VIEW MODE: normal display ----
                <>
                  <input
                    type="checkbox"
                    checked={todo.completed}
                    onChange={() => handleToggle(todo.id)}
                  />
                  <span
                    style={{
                      ...styles.todoText,
                      textDecoration: todo.completed ? "line-through" : "none",
                      color: todo.completed ? "#888" : "#000",
                    }}
                  >
                    {todo.text}
                  </span>
                  <button
                    onClick={() => handleEditStart(todo)}
                    style={styles.editBtn}
                  >
                    Edit
                  </button>
                  <button
                    onClick={() => handleDelete(todo.id)}
                    style={styles.deleteBtn}
                  >
                    Delete
                  </button>
                </>
              )}
            </li>
          ))}
        </ul>
      )}
 
      <div style={styles.footer}>
        <span>{todos.length} tasks</span>
        <span>{completedCount} completed</span>
      </div>
    </div>
  );
}
